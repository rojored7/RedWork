# RedWork Secret Scanner - Pattern Definitions
# Format: CATEGORY | PATTERN | DESCRIPTION
# Lines starting with # are comments
# Patterns use grep -E (extended regex) syntax

# --- AWS ---
AWS_ACCESS_KEY | AKIA[0-9A-Z]{16} | AWS Access Key ID
AWS_SECRET_KEY | (?i)aws_secret_access_key[[:space:]]*[:=][[:space:]]*['\"][A-Za-z0-9/+=]{40} | AWS Secret Access Key
AWS_ACCOUNT_ID | (?i)aws_account_id[[:space:]]*[:=][[:space:]]*['\"][0-9]{12}['\"] | AWS Account ID

# --- GCP ---
GCP_API_KEY | AIza[0-9A-Za-z_-]{35} | Google Cloud API Key
GCP_SERVICE_ACCOUNT | \"type\":[[:space:]]*\"service_account\" | GCP Service Account JSON

# --- Azure ---
AZURE_STORAGE_KEY | (?i)AccountKey=[A-Za-z0-9+/=]{88} | Azure Storage Account Key
AZURE_CONNECTION_STRING | (?i)DefaultEndpointsProtocol=https;AccountName= | Azure Connection String

# --- Generic API Keys ---
GENERIC_API_KEY | (?i)(api[_-]?key|apikey)[[:space:]]*[:=][[:space:]]*['\"][A-Za-z0-9]{20,}['\"] | Generic API Key assignment
GENERIC_SECRET | (?i)(secret[_-]?key|secretkey)[[:space:]]*[:=][[:space:]]*['\"][A-Za-z0-9]{20,}['\"] | Generic Secret Key
GENERIC_ACCESS_TOKEN | (?i)(access[_-]?token)[[:space:]]*[:=][[:space:]]*['\"][A-Za-z0-9._-]{20,}['\"] | Generic Access Token

# --- Tokens ---
BEARER_TOKEN | (?i)(token|bearer)[[:space:]]*[:=][[:space:]]*['\"][A-Za-z0-9._-]{20,}['\"] | Bearer/Token assignment
JWT_TOKEN | eyJ[A-Za-z0-9_-]*\.eyJ[A-Za-z0-9_-]*\.[A-Za-z0-9_-]* | JWT Token (encoded)

# --- Passwords ---
PASSWORD_ASSIGNMENT | (?i)(password|passwd|pwd)[[:space:]]*[:=][[:space:]]*['\"][^'\"]{8,}['\"] | Password assignment in code
DB_PASSWORD | (?i)(db_password|database_password|db_pass)[[:space:]]*[:=][[:space:]]*['\"][^'\"]{4,}['\"] | Database password

# --- Private Keys ---
RSA_PRIVATE_KEY | -----BEGIN RSA PRIVATE KEY----- | RSA Private Key header
EC_PRIVATE_KEY | -----BEGIN EC PRIVATE KEY----- | EC Private Key header
DSA_PRIVATE_KEY | -----BEGIN DSA PRIVATE KEY----- | DSA Private Key header
GENERIC_PRIVATE_KEY | -----BEGIN PRIVATE KEY----- | Generic Private Key header
OPENSSH_PRIVATE_KEY | -----BEGIN OPENSSH PRIVATE KEY----- | OpenSSH Private Key header
PGP_PRIVATE_KEY | -----BEGIN PGP PRIVATE KEY BLOCK----- | PGP Private Key header

# --- Database URLs ---
POSTGRES_URL | (?i)postgres(ql)?://[^[:space:]]+:[^[:space:]]+@[^[:space:]]+ | PostgreSQL connection with credentials
MYSQL_URL | (?i)mysql://[^[:space:]]+:[^[:space:]]+@[^[:space:]]+ | MySQL connection with credentials
MONGODB_URL | (?i)mongodb(\+srv)?://[^[:space:]]+:[^[:space:]]+@[^[:space:]]+ | MongoDB connection with credentials
REDIS_URL | (?i)redis://:[^[:space:]]+@[^[:space:]]+ | Redis connection with password

# --- GitHub/GitLab ---
GITHUB_TOKEN | gh[pousr]_[A-Za-z0-9_]{36,} | GitHub Personal Access Token
GITHUB_OAUTH | (?i)github[_-]?(oauth|token|secret)[[:space:]]*[:=][[:space:]]*['\"][A-Za-z0-9]{35,}['\"] | GitHub OAuth token
GITLAB_TOKEN | glpat-[A-Za-z0-9_-]{20,} | GitLab Personal Access Token
GITLAB_RUNNER_TOKEN | (?i)gitlab[_-]?runner[_-]?token[[:space:]]*[:=][[:space:]]*['\"][A-Za-z0-9-]{20,}['\"] | GitLab Runner Registration Token

# --- Slack ---
SLACK_TOKEN | xox[bporas]-[0-9A-Za-z-]+ | Slack API Token
SLACK_WEBHOOK | https://hooks\.slack\.com/services/T[A-Z0-9]+/B[A-Z0-9]+/[A-Za-z0-9]+ | Slack Webhook URL

# --- Stripe ---
STRIPE_SECRET_KEY | sk_live_[A-Za-z0-9]{24,} | Stripe Live Secret Key
STRIPE_RESTRICTED_KEY | rk_live_[A-Za-z0-9]{24,} | Stripe Live Restricted Key

# --- Twilio ---
TWILIO_API_KEY | SK[A-Za-z0-9]{32} | Twilio API Key

# --- SendGrid ---
SENDGRID_API_KEY | SG\.[A-Za-z0-9_-]{22,}\.[A-Za-z0-9_-]{43,} | SendGrid API Key

# --- Mailgun ---
MAILGUN_API_KEY | key-[A-Za-z0-9]{32} | Mailgun API Key

# --- npm ---
NPM_TOKEN | (?i)//registry\.npmjs\.org/:_authToken=[A-Za-z0-9-]+ | npm Registry Auth Token

# --- Docker ---
DOCKER_AUTH | (?i)\"auth\":[[:space:]]*\"[A-Za-z0-9+/=]{20,}\" | Docker Registry Auth

# --- SSH ---
SSH_PASSWORD_IN_URL | ssh://[^@]+:[^@]+@[^[:space:]]+ | SSH URL with embedded password

# --- Encryption ---
ENCRYPTION_KEY | (?i)(encryption[_-]?key|encrypt[_-]?key|aes[_-]?key)[[:space:]]*[:=][[:space:]]*['\"][A-Fa-f0-9]{32,}['\"] | Encryption key hex

# --- Environment file content ---
ENV_SECRET | (?i)^[A-Z_]*(SECRET|KEY|TOKEN|PASSWORD|CREDENTIAL|AUTH)[A-Z_]*[[:space:]]*=[[:space:]]*['\"]?[A-Za-z0-9/+=._-]{10,}['\"]? | Secret in env file format

# --- OpenAI ---
OPENAI_API_KEY | sk-[A-Za-z0-9]{32,} | OpenAI API Key

# --- Anthropic ---
ANTHROPIC_API_KEY | sk-ant-[A-Za-z0-9-]{80,} | Anthropic API Key

# --- Supabase ---
SUPABASE_KEY | eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+ | Supabase anon/service key (JWT)

# --- Firebase ---
FIREBASE_KEY | (?i)\"apiKey\":[[:space:]]*\"AIza[0-9A-Za-z_-]{35}\" | Firebase API Key in config

# --- Heroku ---
HEROKU_API_KEY | (?i)heroku[_-]?api[_-]?key[[:space:]]*[:=][[:space:]]*[A-Fa-f0-9-]{36} | Heroku API Key

# --- DigitalOcean ---
DIGITALOCEAN_TOKEN | dop_v1_[A-Fa-f0-9]{64} | DigitalOcean Personal Access Token

# --- Datadog ---
DATADOG_API_KEY | (?i)dd[_-]?api[_-]?key[[:space:]]*[:=][[:space:]]*['\"][A-Fa-f0-9]{32}['\"] | Datadog API Key

# --- Shopify ---
SHOPIFY_TOKEN | shpat_[A-Fa-f0-9]{32} | Shopify Admin API Token

# --- Mailchimp ---
MAILCHIMP_API_KEY | [A-Fa-f0-9]{32}-us[0-9]{1,2} | Mailchimp API Key

# --- Generic Webhook ---
GENERIC_WEBHOOK_SECRET | (?i)webhook[_-]?secret[[:space:]]*[:=][[:space:]]*['\"][A-Za-z0-9]{16,}['\"] | Webhook signing secret
